  <script>

    // Direct link to your Google Sheet data

    const CSV_URL = "https://docs.google.com/spreadsheets/d/1U_-WZn1fGXTPngtlEMYRA8a3bUIDMhlxvppxME6AsWQ/export?format=csv&gid=0";


    let lastToken1 = null;

    let lastToken5 = null;

    let audioAllowed = false;


    function enableAudioAndStart() {

      const audio = document.getElementById("chime-sound");

      audio.play().then(() => {

        audio.pause();

        audio.currentTime = 0;

        audioAllowed = true;

        document.getElementById("audio-overlay").style.display = "none";

      }).catch(() => {

        audioAllowed = true;

        document.getElementById("audio-overlay").style.display = "none";

      });

      fetchQueueData();

      setInterval(fetchQueueData, 5000); // Checks every 5 seconds

    }


    async function fetchQueueData() {

      try {

        const response = await fetch(CSV_URL + "&t=" + new Date().getTime());

        const text = await response.text();

        const rows = parseCSV(text);


        const token1 = rows[6] ? rows[6][1]?.trim() : "--";

        const token5 = rows[6] ? rows[6][3]?.trim() : "--";

        const next1 = rows[7] ? rows[7][1]?.trim() : "--";

        const next5 = rows[7] ? rows[7][3]?.trim() : "--";


        if (lastToken1 !== null && token1 !== lastToken1 && token1 !== "--") {

          playAlert("card-opd1");

        }

        if (lastToken5 !== null && token5 !== lastToken5 && token5 !== "--") {

          playAlert("card-opd5");

        }


        lastToken1 = token1;

        lastToken5 = token5;


        document.getElementById("token-opd1").innerText = token1;

        document.getElementById("token-opd5").innerText = token5;

        document.getElementById("next-opd1").innerText = next1;

        document.getElementById("next-opd5").innerText = next5;

      } catch (err) {

        console.error("Queue sync error:", err);

      }

    }


    function playAlert(cardId) {

      if (audioAllowed) {

        const audio = document.getElementById("chime-sound");

        audio.currentTime = 0;

        audio.play().catch(e => console.log("Audio waiting for user click", e));

      }

      const card = document.getElementById(cardId);

      if (card) {

        card.classList.add("highlight");

        setTimeout(() => card.classList.remove("highlight"), 2000);

      }

    }


    function parseCSV(str) {

      const arr = [];

      let quote = false;

      let row = 0, col = 0;

      for (let c = 0; c < str.length; c++) {

        let cc = str[c], nc = str[c+1];

        arr[row] = arr[row] || [];

        arr[row][col] = arr[row][col] || '';

        if (cc === '"' && quote && nc === '"') { arr[row][col] += '"'; ++c; continue; }

        if (cc === '"') { quote = !quote; continue; }

        if (cc === ',' && !quote) { ++col; continue; }

        if (cc === '\r' && nc === '\n' && !quote) { ++row; col = 0; ++c; continue; }

        if (cc === '\n' && !quote) { ++row; col = 0; continue; }

        if (cc === '\r' && !quote) { ++row; col = 0; continue; }

        arr[row][col] += cc;

      }

      return arr;

    }
